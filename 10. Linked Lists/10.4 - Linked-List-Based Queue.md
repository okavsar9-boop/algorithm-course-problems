Implement a Queue class using a singly linked list.

It should support the following methods:


push(v):
    Adds a value v at the back of the queue.

pop():
    Removes the value at the front of the queue and returns its value.
    If the queue is empty, returns None.

peek():
    Returns the value at the front of the queue without removing it.
    If the queue is empty, returns None.

size():
    Returns the number of elements in the queue.

empty():
    Returns True if the queue is empty, False otherwise.
Examples:


Example 1:
queue = Queue()
queue.push(1)    # Queue is now: 1
queue.push(2)    # Queue is now: 2->1
queue.push(3)    # Queue is now: 3->2->1
queue.peek()     # Returns 1
queue.size()     # Returns 3
queue.empty()    # Returns False
queue.pop()      # Returns 1, queue is now: 3->2
queue.pop()      # Returns 2, queue is now: 3
queue.pop()      # Returns 3, queue is now empty
queue.empty()    # Returns True

Example 2:
queue = Queue()
queue.pop()      # Returns None (empty queue)
queue.peek()     # Returns None (empty queue)
queue.size()     # Returns 0
queue.empty()    # Returns True
Constraints:

If your language is typed, you can either make the type of the queue elements generic, or use integers.
All the methods should take O(1) time if the elements are integers.
The queue can contain up to 10^5 elements.

Problem 34.4 - Linked-List-Based Queue: Solution
Python
We can use a singly linked list implementation where we maintain both a head and a tail pointer. The key insight is to use different ends of the linked list for different operations:

The back of the linked list (tail) is used for push()
The front of the linked list (head) is used for pop() and peek()
To satisfy the specified interface, the method that is typically called push_back() in a linked list gets renamed to push(), and pop_front() gets renamed to pop(). The peek() method simply checks the value at the head of the linked list.

To support the size() method in constant time, we keep track of the number of elements in the linked list in a variable.

The main challenge is maintaining both head and tail pointers correctly, especially when the queue becomes empty or when there's only one element.

Here is the implementation:

class Node:
  def __init__(self, val):
    self.val = val
    self.next = None

class LinkedListQueue:
  def __init__(self):
    self.head = None
    self.tail = None
    self._size = 0

  def empty(self):
    return not self.head

  def size(self):
    return self._size

  def push(self, val):
    new = Node(val)
    if self.tail:
      self.tail.next = new
    self.tail = new
    if not self.head:
      self.head = new
    self._size += 1

  def pop(self):
    if self.empty():
      return None
    val = self.head.val
    self.head = self.head.next
    if not self.head:
      self.tail = None
    self._size -= 1
    return val

class Node {
  constructor(val = 0, next = null) {
    this.val = val;
    this.next = next;
  }
}

class LinkedListQueue {
  constructor() {
    this.head = null;
    this.tail = null;
    this._size = 0;
  }

  empty() {
    return !this.head;
  }

  size() {
    return this._size;
  }

  push(val) {
    const newNode = new Node(val);
    if (this.tail) {
      this.tail.next = newNode;
    }
    this.tail = newNode;
    if (!this.head) {
      this.head = newNode;
    }
    this._size++;
  }

  pop() {
    if (this.empty()) {
      return null;
    }
    const val = this.head.val;
    this.head = this.head.next;
    if (!this.head) {
      this.tail = null;
    }
    this._size--;
    return val;
  }
}

class Node {
  int val;
  Node next;

  Node(int x) {
    val = x;
    next = null;
  }
}

class LinkedListQueue {
  private Node head;
  private Node tail;
  private int size;

  public LinkedListQueue() {
    head = null;
    tail = null;
    size = 0;
  }

  public boolean empty() {
    return head == null;
  }

  public int size() {
    return size;
  }

  public void push(int val) {
    Node newNode = new Node(val);
    if (tail != null) {
      tail.next = newNode;
    }
    tail = newNode;
    if (head == null) {
      head = newNode;
    }
    size++;
  }

  public Integer pop() {
    if (empty()) {
      return null;
    }
    Integer val = head.val;
    head = head.next;
    if (head == null) {
      tail = null;
    }
    size--;
    return val;
  }
}

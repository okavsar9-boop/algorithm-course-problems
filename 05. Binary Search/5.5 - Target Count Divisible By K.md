Given a sorted array of integers, arr, a target value, target, and a positive integer, k, return whether the number of occurrences of the target in the array is a multiple of k.


Example 1: arr = [1, 2, 2, 2, 2, 2, 2, 3], target = 2, k = 3
Output: True. 2 occurs 6 times, which is a multiple of 3.

Example 2: arr = [1, 2, 2, 2, 2, 2, 2, 3], target = 2, k = 4
Output: False. 2 occurs 6 times, which is not a multiple of 4.

Example 3: arr = [1, 2, 2, 2, 2, 2, 2, 3], target = 4, k = 3
Output: True. 4 occurs 0 times, and 0 is a multiple of any number.
Constraints:

1 ≤ arr.length ≤ 10^6
-10^9 ≤ arr[i], target ≤ 10^9
1 ≤ k ≤ 10^6
arr is sorted in ascending order

The key is to find the first and last occurrence of the target, first and last. If present, the number of occurrences of the target is last - first + 1. We can check if this number is multiple of k.

If the target is in the array, we can find first and last with a binary search for each, using the transition point recipe twice:

once where we define the is_before as anything < target and returning r,
once where we define the is_before as anything <= target and returning l
For reference, here is what the transition point recipe looks like:

transition_point_recipe()
  define is_before(val) to return whether val is 'before'
  initialize l and r to the first and last values in the range
  handle edge cases:
    - the range is empty
    - l is 'after'  (the whole range is 'after')
    - r is 'before' (the whole range is 'before')

  while l and r are not next to each other (r - l > 1)
    mid = (l + r) / 2
    if is_before(mid)
      l = mid
    else
      r = mid

  return l (the last 'before'), r (the first 'after'), or something else,
         depending on the problem
Here is the implementation:

def target_count_divisible_by_k(arr, target, k):

  # Returns -1 if target is not in arr, or the first index of target in arr
  # otherwise.
  def binary_search_first():

    def is_before(i):
      return arr[i] < target

    l, r = 0, len(arr) - 1
    if arr[l] > target or arr[r] < target:
      return -1
    if arr[l] == target:
      return l

    while r - l > 1:
      mid = (l + r) // 2
      if is_before(mid):
        l = mid
      else:
        r = mid

    if arr[r] == target:
      return r
    return -1

  # Assumes target is in arr. Returns the last index of target in arr.
  def binary_search_last():
    
    def is_before(i):
      return arr[i] <= target

    l, r = 0, len(arr) - 1
    if arr[r] == target:
      return r

    while r - l > 1:
      mid = (l + r) // 2
      if is_before(mid):
        l = mid
      else:
        r = mid
    return l

  first = binary_search_first()
  if first == -1:
    return True  # 0 is a multiple of any number.
  last = binary_search_last()
  count = last - first + 1
  return count % k == 0
Depending on whether your language supports functions as first class citizens, you could write one binary search and pass two unique is_before lambdas to the binary search. Returning both pointers from the function would allow you to grab the one you cared about in each run of the algorithm.

function targetCountDivisibleByK(arr, target, k) {
  // Returns -1 if target is not in arr, or the first index of target in arr otherwise.
  function binarySearchFirst() {
    const isBefore = (i) => arr[i] < target;

    let l = 0,
      r = arr.length - 1;
    if (arr[l] > target || arr[r] < target) {
      return -1;
    }
    if (arr[l] === target) {
      return l;
    }

    while (r - l > 1) {
      const mid = Math.floor((l + r) / 2);
      if (isBefore(mid)) {
        l = mid;
      } else {
        r = mid;
      }
    }

    if (arr[r] === target) {
      return r;
    }
    return -1;
  }

  // Assumes target is in arr. Returns the last index of target in arr.
  function binarySearchLast() {
    const isBefore = (i) => arr[i] <= target;

    let l = 0,
      r = arr.length - 1;
    if (arr[r] === target) {
      return r;
    }

    while (r - l > 1) {
      const mid = Math.floor((l + r) / 2);
      if (isBefore(mid)) {
        l = mid;
      } else {
        r = mid;
      }
    }
    return l;
  }

  const first = binarySearchFirst();
  if (first === -1) {
    return true; // 0 is a multiple of any number.
  }
  const last = binarySearchLast();
  const count = last - first + 1;
  return count % k === 0;
}

class TargetCountDivisibleByK {
  // Returns -1 if target is not in arr, or the first index of target in arr
  // otherwise.
  private int binarySearchFirst(int[] arr, int target) {
    int l = 0, r = arr.length - 1;
    if (arr[l] > target || arr[r] < target) {
      return -1;
    }
    if (arr[l] == target) {
      return l;
    }

    while (r - l > 1) {
      int mid = (l + r) / 2;
      if (arr[mid] < target) { // 'before' range: < target
        l = mid;
      } else {
        r = mid;
      }
    }

    if (arr[r] == target) {
      return r;
    }
    return -1;
  }

  // Assumes target is in arr. Returns the last index of target in arr.
  private int binarySearchLast(int[] arr, int target) {
    int l = 0, r = arr.length - 1;
    if (arr[r] == target) {
      return r;
    }

    while (r - l > 1) {
      int mid = (l + r) / 2;
      if (arr[mid] <= target) { // 'before' range: <= target
        l = mid;
      } else {
        r = mid;
      }
    }
    return l;
  }

  public boolean solve(int[] arr, int target, int k) {
    int first = binarySearchFirst(arr, target);
    if (first == -1) {
      return true; // 0 is a multiple of any number.
    }
    int last = binarySearchLast(arr, target);
    int count = last - first + 1;
    return count % k == 0;
  }
}
